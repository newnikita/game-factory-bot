STYLE GUIDE: как делать красивые игры на HTML5 Canvas

Справочный материал. Содержит готовые реализации обязательных визуальных приёмов: частицы, свечение, параллакс, тряска, вспышки, попапы, HUD, плавность, процедурный звук. Адаптируй код под механику своей игры.

2. Система частиц — взрыв, попадание, сбор предметов


const particles = [];

function burst(x, y, color, count = 18, speed = 4) {
    for (let i = 0; i < count; i++) {
        const a = Math.random() * Math.PI * 2;
        const v = speed * (0.4 + Math.random() * 0.8);
        particles.push({
            x: x, y: y,
            vx: Math.cos(a) * v,
            vy: Math.sin(a) * v,
            life: 1,                              // 1 → 0
            decay: 0.015 + Math.random() * 0.02,
            size: 2 + Math.random() * 3,
            color: color
        });
    }
}

function updateParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt * 60;
        p.y += p.vy * dt * 60;
        p.vx *= 0.97;                             // трение
        p.vy *= 0.97;
        p.life -= p.decay * dt * 60;
        if (p.life <= 0) particles.splice(i, 1);
    }
}

function drawParticles(ctx) {
    ctx.save();
    ctx.globalCompositeOperation = 'lighter';     // аддитивное смешивание = сочное свечение
    for (const p of particles) {
        ctx.globalAlpha = Math.max(0, p.life);
        ctx.fillStyle = p.color;
        ctx.beginPath();
        ctx.arc(p.x, p.y, Math.max(0.5, p.size * p.life), 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.restore();
}

// Вызовы: burst(enemy.x, enemy.y, PAL.accent2, 24)  — смерть врага
//         burst(star.x, star.y, PAL.gold, 12, 3)    — сбор звезды
//         burst(player.x, player.y, PAL.accent, 30, 6) — гибель игрока

function drawGlow(ctx, drawShape, color, blur = 18) {
    ctx.save();
    ctx.shadowColor = color;
    ctx.shadowBlur = blur;
    drawShape();                                   // сам объект: fillRect / arc+fill / ...
    ctx.restore();
}

5. Вспышка попадания и score-попапы


let flash = 0;

function addFlash(alpha = 0.35) {
    if (alpha > flash) flash = alpha;
}

function drawFlash(ctx, w, h) {
    if (flash <= 0.02) { flash = 0; return; }
    ctx.fillStyle = 'rgba(255,255,255,' + flash.toFixed(3) + ')';
    ctx.fillRect(0, 0, w, h);
    flash *= 0.84;
}

// Попапы очков: +100 всплывает и улетает
const popups = [];

function addPopup(x, y, text, color) {
    popups.push({ x: x, y: y, text: text, color: color || PAL.gold, life: 1 });
}

function drawPopups(ctx, dt) {
    ctx.save();
    ctx.font = 'bold 20px Arial';
    ctx.textAlign = 'center';
    ctx.shadowColor = 'rgba(0,0,0,0.7)';
    ctx.shadowBlur = 6;
    for (let i = popups.length - 1; i >= 0; i--) {
        const p = popups[i];
        p.y -= 45 * dt;
        p.life -= 1.4 * dt;
        if (p.life <= 0) { popups.splice(i, 1); continue; }
        ctx.globalAlpha = Math.max(0, p.life);
        ctx.fillStyle = p.color;
        ctx.fillText(p.text, p.x, p.y);
    }
    ctx.restore();
}

6. Параллакс-фон: два слоя звёзд (глубина)


function makeStars(count, speedMin, speedMax) {
    const arr = [];
    for (let i = 0; i < count; i++) {
        arr.push({
            x: Math.random(),                     // нормализовано 0..1
            y: Math.random(),
            s: 1 + Math.random() * 2,
            v: speedMin + Math.random() * (speedMax - speedMin),
            a: 0.3 + Math.random() * 0.7
        });
    }
    return arr;
}

const starsFar = makeStars(70, 0.004, 0.010);     // дальний слой: мелкие, тусклые, медленные
const starsNear = makeStars(25, 0.02, 0.04);      // ближний: крупные, яркие, быстрые

function drawParallax(ctx, w, h, t) {             // t — время в секундах
    ctx.save();
    for (const layer of [starsFar, starsNear]) {
        for (const st of layer) {
            const y = ((st.y + t * st.v) % 1) * h;
            ctx.globalAlpha = st.a;
            ctx.fillStyle = '#cfe9ff';
            ctx.beginPath();
            ctx.arc(st.x * w, y, st.s, 0, Math.PI * 2);
            ctx.fill();
        }
    }
    ctx.restore();
}

7. Плавность: delta time и lerp


let lastT = performance.now();

function loop(now) {
    const dt = Math.min(0.05, (now - lastT) / 1000);   // секунды, потолок 50мс
    lastT = now;
    update(dt);
    render();
    requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

8. Стилизованные экраны меню и Game Over + кнопки на canvas


function drawTitle(ctx, text, x, y, size) {
    const maxW = Math.max(120, (ctx.canvas && ctx.canvas.width ? ctx.canvas.width : 480) - 40);
    ctx.save();
    let s = size;
    ctx.font = 'bold ' + s + 'px Arial';
    while (ctx.measureText(text).width > maxW && s > 14) {
        s = Math.max(14, s - 2);
        ctx.font = 'bold ' + s + 'px Arial';
    }
    const g = ctx.createLinearGradient(x, y - s, x, y);
    g.addColorStop(0, '#ffffff');
    g.addColorStop(1, PAL.accent);
    ctx.textAlign = 'center';
    ctx.shadowColor = PAL.accent;
    ctx.shadowBlur = 25;
    ctx.fillStyle = g;
    ctx.fillText(text, x, y);
    ctx.restore();
}

function drawButton(ctx, btn, pressed) {
    ctx.save();
    ctx.translate(btn.x, btn.y);
    const w = btn.w, h = btn.h, r = 14;
    ctx.fillStyle = pressed ? '#00b8cc' : PAL.accent;
    ctx.shadowColor = PAL.accent;
    ctx.shadowBlur = pressed ? 8 : 20;
    ctx.beginPath();
    if (ctx.roundRect) ctx.roundRect(-w / 2, -h / 2, w, h, r);
    else ctx.rect(-w / 2, -h / 2, w, h);
    ctx.fill();
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#04121a';
    ctx.font = 'bold 22px Arial';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText(btn.label, 0, 1);
    ctx.restore();
}

8.1. Жёсткие правила геймплея: управление, урон, финал


// 1) Управление обязано быть осмысленным. Аркада/платформер/шутер: у игрока есть движение
//    влево/вправо (стрелки/WASD + экранные зоны ◀ ▶ на canvas) и действие (прыжок/огонь:
//    Пробел, тап, кнопка ▲ на canvas). «Только прыжок на месте» — лишь если жанр явно
//    это требует, иначе игрок воспринимает игру как сломанную.
// 2) Урон от врага обязателен: контакт сбоку/снизу снимает жизнь или завершает игру.
//    Стомп сверху/попадание сверху — награда. «Враг проходит сквозь игрока» — баг.
// 3) Финалы — достижимые. Нужны И достижимое условие Game Over (жизни/время/падение —
//    проверь, что путь к нему возможен: если земля везде, падение не убьёт), И условие
//    победы (счёт/уровень/цель). Цель в HUD без события победы — баг. Мёртвые if-ы
//    Game Over — баг.
// 4) После урона: 0.3-1.5 сек неуязвимости с миганием, отбрасывание, звук, popup «-1».
//    После стомпа: короткая лазейка 0.2-0.35 сек, чтобы не получить урон вылетом из врага.
// 5) На узких экранах заголовки обязаны влезать: подбирай кегль через measureText
//    (см. drawTitle в разделе 8), фиксированный 60px — обрезает крайние буквы.

9. HUD: полупрозрачная панель сверху


function drawHUD(ctx, score, lives, w) {
    ctx.save();
    ctx.fillStyle = 'rgba(0,0,0,0.35)';
    ctx.fillRect(0, 0, w, 46);
    ctx.font = 'bold 22px Arial';
    ctx.textBaseline = 'top';
    ctx.shadowColor = 'rgba(0,0,0,0.7)';
    ctx.shadowBlur = 4;
    ctx.fillStyle = PAL.gold;
    ctx.textAlign = 'left';
    ctx.fillText('\u2605 ' + score, 14, 12);
    ctx.fillStyle = PAL.accent2;
    ctx.textAlign = 'right';
    ctx.fillText('\u2665'.repeat(Math.max(0, lives)), w - 14, 12);
    ctx.restore();
}

10. Процедурный звук WebAudio (опционально, безопасно)


let AC = null;

function initAudio() {          // вызвать в обработчике ПЕРВОГО жеста (клик по кнопке меню)
    try {
        AC = AC || new (window.AudioContext || window.webkitAudioContext)();
        if (AC.state === 'suspended') AC.resume();
    } catch (e) { AC = null; }
}

function beep(freq, dur, type, vol) {
    if (!AC) return;
    try {
        const o = AC.createOscillator();
        const g = AC.createGain();
        o.type = type || 'square';
        o.frequency.value = freq;
        g.gain.setValueAtTime(vol || 0.04, AC.currentTime);
        g.gain.exponentialRampToValueAtTime(0.0001, AC.currentTime + (dur || 0.08));
        o.connect(g);
        g.connect(AC.destination);
        o.start();
        o.stop(AC.currentTime + (dur || 0.08));
    } catch (e) {}
}

// beep(220, 0.25, 'sawtooth', 0.05) — взрыв
// beep(1320, 0.06, 'sine')          — сбор предмета
// Всё обёрнуто в try/catch: при любой ошибке — просто тишина.

Чек-лист: игра выглядит «дорого», если

  • есть тёмная подложка, любой текст читается поверх фона;
  • палитра ограничена 2-3 акцентами, а не радуга;
  • частицы минимум при двух событиях (попадание + сбор/смерть);
  • свечение на игроке и снарядах, вспышка при уроне, тряска экрана;
  • фон живой: параллакс 2+ слоёв или движущиеся фоновые частицы;
  • движение через dt (без рывков), UI появляется с fade;
  • меню и Game Over стилизованы: градиентный заголовок, свечение, акцентная кнопка;
  • score-попапы при очках, HUD на полупрозрачной панели.
